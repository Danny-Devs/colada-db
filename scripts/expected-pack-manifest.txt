# Adding a published file is a DELIBERATE ACT. If a build change adds an entry
# here, the question to answer before editing this file is not "how do I make CI

#   pnpm pack --pack-destination /tmp && \
#     tar -tzf /tmp/colada-db-*.tgz | sed 's|^package/||' | LC_ALL=C sort
